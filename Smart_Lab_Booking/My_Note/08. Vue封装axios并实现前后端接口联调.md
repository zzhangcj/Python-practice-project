### utils/auth.js

```javascript
//  存储token、存储用户信息、获取token、用户信息，删除token、用户信息这些作用
const TOKEN_KEY = 'token'
const USER_KEY = 'userInfo'

export function getToken() {
  return localStorage.getItem(TOKEN_KEY)
}

export function setToken(token) {
  localStorage.setItem(TOKEN_KEY, token)
}

export function removeToken() {
  localStorage.removeItem(TOKEN_KEY)
}

export function getUserInfo() {
  const str = localStorage.getItem(USER_KEY)
  return str ? JSON.parse(str) : null
}

export function setUserInfo(userInfo) {
  localStorage.setItem(USER_KEY, JSON.stringify(userInfo))
}

export function removeUserInfo() {
  localStorage.removeItem(USER_KEY)
}

// 登出
export function logout() {
  removeToken()
  removeUserInfo()
}
```

### utils/user.js

```javascript
// 仅供页面  setup 函数使用
import { ref } from 'vue'
import { getUserInfo, setToken, setUserInfo } from './auth'

const userInfo = ref(getUserInfo()) // 响应式对象 可以随时更新最新的值

export function useUser() {
  function saveLoginData(data) {
    setToken(data.token)
    setUserInfo(data.user)
    userInfo.value = data.user
  }

  function updateUser(user) {
    setUserInfo(user)
    userInfo.value = data.user
  }

  function reloadUser() {
    userInfo.value = getUserInfo()
  }

  return {
    userInfo,
    saveLoginData,
    updateUser,
    reloadUser
  }
}
```

**修正：updateUser 的赋值对象**

> 上方 `updateUser(user)` 中的 `userInfo.value = data.user` 是笔误。该函数接收的是 `user`，作用域内没有定义 `data`，调用时会抛出 `ReferenceError: data is not defined`。请将该函数替换为：

```javascript
function updateUser(user) {
  setUserInfo(user)
  userInfo.value = user
}
```

>这样，本地存储与页面使用的响应式状态会更新为同一个用户对象。`saveLoginData(data)` 接收的是包含 `token` 和 `user` 的登录数据，其中的 `data.user` 保持不变。

### utils/request.js

```javascript
import axios from 'axios'
import { getToken, logout } from './auth'
import { ElMessage } from 'element-plus'
import router from '@/router'

const service = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL,
  timeout: 30000 //30s
})

// 请求拦截器的配置
service.interceptors.request.use(
  (config) => {
    const token = getToken()
    if (token) {
      config.headers.Authorization = `Bearer ${token}`
    }
    return config
  },
  (err) => Promise.reject(err)
)

// 返回的拦截器配置
service.interceptors.response.use(
  (res) => {
    const data = res.data
    if (data.code !== 200) {
      ElMessage.error(data.message || '请求失败')
      if (data.code === 401) {
        logout()
        router.push('/login')
        return Promise.reject(data)
      }
    }
    return data
  },
  (err) => {
    if (err.response) {
      const httpStatusCode = err.response.status
      if (httpStatusCode === 401) {
        ElMessage.error('登录状态已失效，请重新登录')
        logout()
        router.push('/login')
      } else {
        ElMessage.error(err.response.data?.message || '请求失败')
      }
    } else {
      // 后端没启动，网络异常了
      ElMessage.error('网络异常，请检查后端服务')
    }
    return Promise.reject(err)
  }
)

export default service
```

# 解析

## 1、这一章的需求场景

上一章我们完成后端部分：
统一所有接口返回格式为 `{code, message, data}`，并且写好了全局异常处理器。
后端只是规定好了**返回数据长什么样**，但是前端还没有配套逻辑去接收、识别这套格式。

本章目标：
**编写前端代码，打通完整的登录鉴权整条链路，和后端的规范配对，让登录功能闭环**


```
页面发起请求
    ↓
axios 请求拦截器 → 自动带上 token（Authorization: Bearer xxx）
    ↓
后端接口校验，返回 {code, message, data}
    ↓
axios 响应拦截器
    ├─ code == 200 → 将 data 给到页面，正常渲染
    ├─ code == 401 → 清空登录信息，跳转登录页
    └─ 其他情况 / 网络失败 → 弹出错误提示
```

关键点是：**昨天后端定了规矩（响应格式），今天前端按这个规矩办事（解析 code、统一处理错误）**。两张拼图合上，登录功能才算真正闭环


## 2、后端配置跨域

```python
from starlette.middleware.cors import CORSMiddleware

origins = [
	"http://localhost:5173",
	"http://127.0.0.1:5173",
]

app.add_middleware(
	CORSMiddleware,
	allow_origins=origins,        # 允许的前端源，不要直接写 ["*"]
	allow_credentials=True,       # ✅ 关键：允许前端携带 Authorization token
	allow_methods=["*"],          # 允许所有请求方法 GET POST PUT DELETE OPTIONS
	allow_headers=["*"],          # 允许所有请求头（包含Authorization）
)
```

✅ **需求场景**

浏览器有**同源策略**这个安全规则： 如果前端页面地址、后端接口地址，**协议、域名、端口有任意一个不一样，就属于跨域**，浏览器会拦住前端发给后端的请求。

我们现在：
- Vue 前端：`http://127.0.0.1:5173`
- FastAPI 后端：`http://127.0.0.1:8000` 端口不一样 → 属于跨域。 
- 如果不做跨域配置，浏览器直接拦截请求，前端连登录接口都发不出去，后面的 token 鉴权、所有接口调用全部无法进行


📝**代码解析**

我们在后端添加 `CORSMiddleware`（跨域中间件），作用就是
后端告诉浏览器：**我信任这两个前端地址，允许它跨域访问我**。

1. `origins`：写清楚允许哪几个前端地址访问后端，这里是只放行我们本地的 5173
>不推荐直接写`["*"]`（`*`代表允许所有来源，而且`*`和`allow_credentials=True`不能同时生效，带 token 的请求会直接失败，这就是注释说不要写`*`的原因）

2. `allow_credentials=True`：重中之重！开启**允许前端携带凭证**，我们的 `Authorization: Bearer token` 身份凭证就是在这里放行，**不开这个，前端带的 token 会被浏览器丢掉，鉴权直接失效**

3. `allow_methods=["*"]`：允许前端使用 GET、POST、PUT、DELETE 等所有请求方式（  *  表示都可以）

4. `allow_headers=["*"]`：允许前端传递所有请求头，包含我们放 token 的`Authorization`请求头

### utils/auth.js


✅ **需求场景**

登录成功之后，后端返回了 token 和用户信息。
如果只放在页面普通变量里，**页面一刷新，数据直接消失，用户就要重新登录**。
我们需要一个地方，把 token、用户信息持久保存到浏览器里，页面刷新、浏览器关掉重开之后，数据还在，这就写了 auth.js 封装 localStorage 的工具函数。

📝**必要知识点/代码解析**

1. `localStorage` 是什么：浏览器本地存储空间，**存在用户浏览器里，关闭页面、重启浏览器之后数据还在**。
    - 对比：普通 `ref`、变量，页面一刷新直接消失。
    - 👉 这就是为什么登录成功后要存在这里：刷新页面不用重新登录。
2. 只能存**字符串**！！（这个是高频踩坑点）
    - 存对象 `userInfo`：要用 `JSON.stringify()` 转成字符串再存
    - 取对象：要用 `JSON.parse()` 转回 js 对象
    - 存简单字符串 token，不用转，直接放
3. 四个基础 API（你代码里全部用到了）
    - `setItem(key,值)`：存
    - `getItem(key)`：取
    - `removeItem(key)`：删除
    - 配套：`logout` 就是一次性删掉 token 和用户信息，用于退出登录
4. 区分：`TOKEN_KEY / USER_KEY` 是给存储起名字，相当于给柜子的两个抽屉贴标签，防止存的数据互相覆盖。

> 一句话理解 auth.js：封装一套函数，专门统一管理这两个抽屉的读写，页面不用到处写 `localStorage`。

#### API

API = **Application Programming Interface，应用程序编程接口** 

1. **浏览器内置 API（现在这个）** 浏览器给 JS 提供的**可以直接调用的函数**：localStorage、alert、fetch ✅ 直接在前端代码调用，不用联网 
>	例子：`localStorage.setItem()`,`localStorage.removeItem()`

2. **后端接口 API（你 FastAPI 写的接口，/api/auth/login）** 通过 HTTP 网络请求访问的接口，前后端交互用的，就是我们整条登录链路调用的接口

3. **第三方服务 API（大模型、短信、OSS 存储）** 远程线上服务，需要发 http 请求，往往搭配 apikey 作为身份凭证访问


API 是统称：**凡是给代码调用的功能入口，都叫 API,或者说接口**
**“接口” 是日常口语叫法；API 是它的正式学名。大部分场景二者可以混用**

### utils/user.js

```javascript
// 仅供页面  setup 函数使用
import { ref } from 'vue'
import { getUserInfo, setToken, setUserInfo } from './auth'

const userInfo = ref(getUserInfo()) // 响应式对象 可以随时更新最新的值

export function useUser() {
  function saveLoginData(data) {
    setToken(data.token)
    setUserInfo(data.user)
    userInfo.value = data.user
  }

  function updateUser(user) {
    setUserInfo(user)
    userInfo.value = data.user
  }

  function reloadUser() {
    userInfo.value = getUserInfo()
  }

  return {
    userInfo,
    saveLoginData,
    updateUser,
    reloadUser
  }
}
```

✅ 需求场景 

- 前面auth.js 只解决了把用户信息、token 存进 localStorage 的问题。 
- 而Vue 页面想要进行渲染，比如展示用户名（欢迎 xxx），需要用到 Vue 的响应式变量 `ref`；
- 但是 localStorage 里面的数据和vue没联系，vue不知道里面的数据变没变，就**不会自动渲染到页面上**。 
- 所以我们需要**修改存储的时候，同步更新 ref**；ref 变了，Vue 页面才自动渲染

📝代码解析

1. 初始化逻辑：`const userInfo = ref(getUserInfo())` 页面刷新的时候，一开始就从 localStorage 读取已经保存好的用户信息，给响应式变量赋初始值，实现刷新页面用户名还能正常展示

2. 封装函数作用
- `saveLoginData(data)`：**登录成功时调用**。调用 auth 方法把 token、用户信息存入 localStorage，同时更新`userInfo.value`，存储和页面状态同步。

- `updateUser(user)`：**修改用户资料时调用**。更新 localStorage 里的用户数据，同步更新响应式变量，页面立刻展示最新用户信息。

- `reloadUser()`：**重新从 localStorage 读取用户信息**，覆盖当前 ref 的值，用来手动同步最新登录状态。

-  组合式函数 `useUser()` 把用户相关的状态和方法统一封装导出，页面只需要导入`useUser()`就可以直接使用，多处页面复用，不用重复写 ref、读取存储的逻辑。

#### axios插件
为什么需要这个插件呢，它是做什么用的？
```
类比寄送快递的例子

页面调用 loginApi（寄快递）
→ 进入【请求拦截器】发货检查站：检查有没有 token，贴上身份头
→ 发给后端
→ 后端处理完，返回包裹（{code,message,data}）
→ 包裹回来，进入【响应拦截器】收货检查站：判断 code，处理报错 / 401 跳转
→ 处理完毕，最后把结果给到页面代码
```

（1）我的理解
意思是这样的，这一趟流程我们需要这两个检查站，但原本的传输链上没有
**所以需要我们加装新建，于是需要axios给我们这个介入这个传输链的机会**

（2）豆包解析
原本浏览器发请求、收响应这条传输路线，是光秃秃直接走的，**没有地方让我们统一加检查逻辑**。 axios 给我们开放了两个 “介入点（钩子）”：

1. 请求发出去之前
2. 响应拿到之后、页面拿到数据之前

我们就利用这两个介入点，**自己写代码，加装这两个检查站**。
**检查站不是 axios 自带的，是我们自己写的；axios 只是提供了能插进去的位置**

utils/request.js
✅ 需求场景 
auth.py解决存储，user.py确保前端自动渲染跟上
这里就是需要利用axios，建造这两个检查站了

- 请求拦截器（发货检查站）：自动读取 token，按规范放到请求头里发给后端
- 响应拦截器（收货检查站）：统一处理返回结果，捕获报错；检测到 token 失效就清登录态、跳登录页

📝代码解析

#### axios baseURL 自动拼接路径
```javascript
const service = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL,
  timeout: 30000 //超时时间：30秒，30秒后端没有响应就判定请求超时失败
})
```
`baseURL: import.meta.env.VITE_API_BASE_URL`
这一步，和前面 4. 配置环境变量 相联系

还是送快递这个案例，baseURL 相当于预先填好的**后端驿站地址**
这个驿站的**窗口**（`/auth/login`、`/user/info` 这类路由）是固定不变的；

寄快递的时候：使用**驿站地址 + 窗口名称**，拼出完整目标位置，就可以把请求送到后端对应的窗口
之所以要把驿站地址存在环境变量.env里面，是因为后续这个地址会变，但窗口不会变

**好处** 
① 不用每个接口都写一长串完整地址，减少重复代码； 
② 后端地址更换（本地开发 → 服务器上线），只修改环境变量里的 baseURL，接口代码不用改动

## 4、发送 api 请求 — api/auth.js

```javascript
import request from '@/utils/request'
export function loginApi(data) {
  return request({
    url: '/api/auth/login',
    method: 'post',
    data
  })
}
```

✅ 需求场景 我们已经封装好了底层 `request`（带拦截器、自动带token、统一报错的axios实例），但是 `request` 只是通用的网络工具。 如果直接在 Login.vue 里面写 `request({url,method...})`： 接口地址、请求方式散落在页面里。以后后端接口路径改了、接口调整，要到处找页面改代码，不好维护。

所以我们单独建立 `api` 目录，**按业务模块把接口封装成一个个函数**。 把登录这个网络请求单独抽成 `loginApi`。 页面只需要导入调用 `loginApi(表单数据)`，不用关心url、method。 做到：**接口定义集中管理，视图页面只关心业务逻辑，和网络细节解耦**。

## 5、路由守卫 — router/index.js

在router/index.js修改

```javascript
import { getToken } from '@/utils/auth'

// 路由守卫：验证token
router.beforeEach((to, from) => {
  const token = getToken()

  if (to.path === '/login' || to.path === '/register') {
    return true
  } else {
    return token ? true : '/login'
  }
})
```
✅ 需求场景 前面我们只做到：登录成功有token、存到浏览器。 但是有一个漏洞：用户可以直接在浏览器地址栏手动输入 `/manager/home`，绕过登录页面直接访问主页。

我们需要在每一次切换页面（路由跳转）之前做拦截检查：
- 去登录、注册页：直接放行
- 去其他需要登录的页面：读取本地token，有token才放行；没有token，强制跳回登录页

它的定位：**全局页面访问权限保护**。 和接口层面的JWT鉴权是两层防护：

- 路由守卫：前端层面，拦住页面入口，没token不让进页面（只是前端拦截，只是体验层防护）
- JWT + 后端接口鉴权：后端接口校验token，是真正的安全防线

> 重点笔记区分：路由守卫拦的是**页面访问**；后端JWT校验拦的是**接口请求**。两个配合使用

跨域配置 → 底层request封装（请求/响应拦截） → auth.js本地存储工具 → useUser维护响应式用户状态 → api层封装接口函数 → Login页面交互发起登录、保存凭证 → 路由守卫拦截非法页面访问 → 页面读取userInfo展示信息 从浏览器请求放行、网络请求、凭证持久化、状态响应、页面交互、页面访问权限，整套登录鉴权闭环全部完成。
