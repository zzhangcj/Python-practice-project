### Equipment 模型

新建 `models/equipment.py`。设备属于实验室，用外键关联：

```python
from sqlalchemy.orm import Mapped, mapped_column, relationship
from sqlalchemy import ForeignKey, Integer, String
from app.database import Base
from app.models.lab import Lab

class Equipment(Base):
    __tablename__ = "equipments"
    __table_args__ = {"comment": "实验室设备表"}

    lab_id: Mapped[int] = mapped_column(
        ForeignKey("labs.id"), comment="所属实验室", nullable=False
    )
    name: Mapped[str] = mapped_column(String(50), comment="设备名称", nullable=False)
    description: Mapped[str | None] = mapped_column(
        String(500), comment="说明", nullable=True
    )
    img: Mapped[str | None] = mapped_column(String(200), comment="图片", nullable=True)
    spec: Mapped[str | None] = mapped_column(
        String(100), comment="型号规格", nullable=True
    )
    quantity: Mapped[int] = mapped_column(Integer, comment="数量", default=1)
    status: Mapped[int] = mapped_column(default=1, comment="状态：0-维修，1-正常")

    lab: Mapped[Lab] = relationship()
```

### equipment_service

新建 `services/equipment_service.py`：

```python
from operator import or_
from app.common.exceptions import BusinessException
from app.common.response import PageResponse
from app.models.equipment import Equipment
from app.models.lab import Lab
from app.schemas.equipment import (
    EquipmentResponse,
    EquipmentCreateRequest,
    EquipmentUpdateRequest,
)
from sqlalchemy.orm import Session

def get_equipment_page_list(
    db: Session,
    page: int,
    page_size: int,
    keywords: str | None = None,
    lab_id: int | None = None,
):
    """分页模糊查询列表"""
    # select * from equipment where name like '%计算机%';
    query = db.query(Equipment)
    if lab_id:
        query = query.filter(Equipment.lab_id == lab_id)
    if keywords:
        query = query.filter(Equipment.name.ilike(f"%{keywords}%"))
    total = query.count()
    items = (
        query.order_by(Equipment.id.desc())
        .offset((page - 1) * page_size)
        .limit(page_size)
        .all()
    )
    result = []
    for item in items:
        res = EquipmentResponse.model_validate(item)
        res.lab_name = item.lab.name if item.lab else None
        result.append(res)
    return PageResponse(list=result, total=total)

def create_equipment(db: Session, data: EquipmentCreateRequest):
    """新增"""
    exist = db.query(Equipment).filter(Equipment.name == data.name).first()
    if exist:
        raise BusinessException(message="实验室设备已存在")
    equipment = Equipment(**data.model_dump())
    db.add(equipment)
    db.commit()
    db.refresh(equipment)
    res = EquipmentResponse.model_validate(equipment)
    res.lab_name = equipment.lab.name if equipment.lab else None
    return res

def update_equipment(db: Session, equipment_id: int, data: EquipmentUpdateRequest):
    """更新"""
    equipment = db.query(Equipment).filter(Equipment.id == equipment_id).first()
    if not equipment:
        raise BusinessException(message="实验室设备不存在")
    payload = data.model_dump(exclude_none=True)
    lab_id = payload.get("lab_id", equipment.lab_id)
    name = payload.get("name", equipment.name)
    lab = db.query(Lab).filter(Lab.id == lab_id).first()
    if not lab:
        raise BusinessException(message="实验室不存在")
    exist = (
        db.query(Equipment)
        .filter(
            Equipment.lab_id == lab_id,
            Equipment.name == name,
            Equipment.id != equipment.id,
        )
        .first()
    )
    if exist:
        raise BusinessException(message="同一个实验室不能存在同名的设备")

    for filed, value in payload.items():
        setattr(equipment, filed, value)

    db.commit()
    db.refresh(equipment)
    res = EquipmentResponse.model_validate(equipment)
    res.lab_name = equipment.lab.name if equipment.lab else None
    return EquipmentResponse.model_validate(equipment)

def delete_equipment(db: Session, equipment_id: int):
    """删除"""
    equipment = db.query(Equipment).filter(Equipment.id == equipment_id).first()
    if not equipment:
        raise BusinessException(message="实验室设备不存在")
    db.delete(equipment)
    db.commit()
```

新增、修改都要确认实验室存在。同一个实验室里设备名称不能重复，不同实验室可以都有 RTX4090。

### equipment API

新建 `api/equipment.py`：

```python
from fastapi import APIRouter, Depends
from sqlalchemy.orm import Session
from app.common.response import Response
from app.models.equipment import Equipment
from app.dependencies.auth import get_current_admin
from app.schemas.equipment import (
    EquipmentCreateRequest,
    EquipmentUpdateRequest,
)
from app.services import equipment_service
from app.database import get_db

router = APIRouter(prefix="/equipment", tags=["实验室设备信息接口"])

@router.get("/list")
def get_equipment_list(
    page: int = 1,
    page_size: int = 10,
    keywords: str | None = None,
    lab_id: int | None = None,
    current_user: Equipment = Depends(get_current_admin),
    db: Session = Depends(get_db),
):
    res = equipment_service.get_equipment_page_list(
        db, page, page_size, keywords, lab_id
    )
    return Response.success(data=res)

@router.post("")
def create_equipment(
    data: EquipmentCreateRequest,
    current_user: Equipment = Depends(get_current_admin),
    db: Session = Depends(get_db),
):
    res = equipment_service.create_equipment(db, data)
    return Response.success(data=res)

@router.put("/{equipment_id}")
def update_equipment(
    equipment_id: int,
    data: EquipmentUpdateRequest,
    current_user: Equipment = Depends(get_current_admin),
    db: Session = Depends(get_db),
):
    res = equipment_service.update_equipment(db, equipment_id, data)
    return Response.success(data=res)

@router.delete("/{equipment_id}")
def delete_equipment(
    equipment_id: int,
    current_user: Equipment = Depends(get_current_admin),
    db: Session = Depends(get_db),
):
    equipment_service.delete_equipment(db, equipment_id)
    return Response.success()
```

# 解析

## 1、Equipment 模型
### 外键 + relationship 关联

代码详见上面 Equipment 模型

**1. 需求情景 / 使用动机**

我们要让每个设备都关联一个实验室。
我们要做到两件事：

1. 不能给不存在的实验室添加设备；
2. 查设备的时候，可以直接拿到它所属实验室的名字，不用自己再写复杂 SQL语句来把两张表连起来

**2. 代码逐段解释**

> 关联相关代码提取：
```python
# 外键
lab_id: Mapped[int] = mapped_column(
    ForeignKey("labs.id"), comment="所属实验室", nullable=False
)
# ORM关联导航
lab: Mapped[Lab] = relationship()
```

✅ `ForeignKey("labs.id")` 
向数据库声明外键约束，底层由数据库做数据校验
大白话：告诉数据库，`lab_id` 必须去 `labs` 这张表里找对应的 id

- 数据库这边生效，表里真的有 `lab_id` 这一列，并设置为外键
- 数据库校验效果：
    - 如果你填一个不存在的实验室 id 去新增设备 → 数据库直接拒绝保存
    - 如果你想删掉一个还有设备挂着的实验室 → 默认删不掉，数据库阻止你
- SQLAlchemy 只是帮你把这条规则告诉数据库
- 我们写它的目的：避免出现 “设备归属一个根本不存在的实验室” 这种错误数据

<br>

✅ `lab: Mapped[Lab] = relationship()` 
作用：拿到设备对象后，可以直接通过 `.lab` 拿到对应的实验室整条数据

- 数据库表里看不到这个 `lab`，是ORM 提供的导航属性
- 效果：当你拿到一条设备数据后，可以直接写 `equipment.lab.name`，拿到这个设备所属实验室的名字
- 我们写它的目的：
	- SQLAlchemy 帮你自动去关联查表拿实验室信息
	- 简化代码查询，不用手写 JOIN进行连表查询
- 加载特性：默认懒加载，不访问 `.lab` 就不会额外发起查询

**3. 必要知识点**

1. `ForeignKey`：数据库层面的规则，实实在在存在于数据表
2. `relationship`：只在 Python 代码里生效，数据库没有这一列
3. 关系：多个设备属于同一个实验室（多对一）
4. 外键 设置属性`nullable=False`：新增设备必须选择实验室，不允许设备没有归属
5. 默认懒加载：你不写 `.lab`，它就不去查实验室；你访问 `.lab`，它才额外查一次数据库
6. 外键只是兜底保护，业务上我们还是要先在 service 里判断实验室是否存在，给前端友好提示

### nullable属性
核心一句话：**nullable 控制该字段是否允许为 NULL（空值）**
```python
nullable=False  → 不允许为空（NOT NULL），必填
nullable=True   → 允许为空（DEFAULT，默认值就是 True）
```

### 字段/主键和外键约束

❌ 我的理解：

> A 表的 id；B 表有一模一样的 id 字段，能靠它连 A 表，那 B 里这个字段就叫外键，在A哪里就叫主键

思维误区： 
- 字段 `lab_id` 本身，就只是一个存数字的普通列，它**天然不是外键**
- 是你给这个字段**挂上了外键约束**之后，它才变成 “外键字段”
- 字段是载体，约束是附加在它身上的规则，二者不能混为一谈

<br>

✅ 对照主键帮你对比，一下子就通：
- `labs.id`：只是一个存数字的列；给它加上**主键约束（PRIMARY KEY）** → 它成为主键字段，规则：不能重复、不能为空、唯一标识一行。
>那么，加上主键约束这个操作在哪呢？
>在`utils/database.py`
- `equipments.lab_id`：只是一个存数字的列；给它加上**外键约束（FOREIGN KEY）** → 它成为外键字段，规则：它的值必须在 labs 的主键里面存在。

👉 一句话： **字段是容器，主键 / 外键是贴在容器上的规则标签。** 
>我们口语经常偷懒，说 “这个字段是外键”，是简化说法
严格来说：这个字段**带有外键约束**

- 含义：在整张设备表里面，只要有任意一条设备和本次提交的name相同，直接报错
- 范围：**全表、全局唯一**，不区分实验室
- 特点：逻辑简单；只要名字重复，不管是不是同一个实验室，都不让新增

#### ② update_equipment 更新校验

```python
exist = (
    db.query(Equipment)
    .filter(
        Equipment.lab_id == lab_id,
        Equipment.name == name,
        Equipment.id != equipment.id,
    )
    .first()
)
if exist:
    raise BusinessException(message="同一个实验室不能存在同名的设备")
```

- 过滤条件同时3条：
    1. `Equipment.lab_id == lab_id`：实验室编号等于本次更新后的实验室编号
    2. `Equipment.name == name`：设备名称等于本次更新后的名称
    3. `Equipment.id != equipment.id`：设备 id 不是我现在正在修改的这条设备
       >多个条件默认是 **并且（AND）** 的关系，必须**全部满足**才算匹配
- 含义：去设备表里找一条**别的设备**，它满足
	1. 它的实验室 = 我更新之后要改成的实验室
	2. 它的名字 = 我更新之后要改成的名字
	3. 它不是我自己这条设备
	>只有同时满足这三点，才代表：**有别人和我重名了**，才拦截
- 范围：**单实验室内部联合唯一（lab_id + name）**

**核心关键点：id != equipment.id**

这句话的意思就是，设备 id 不是我现在正在修改的这条设备
如果不写`id != equipment.id`：用户只修改别的字段、不改设备名称时，查询会匹配到自己，误报同名错误，无法保存

看不懂的话，看看下面的前置条件

| 维度      | 新增 create 校验  | 更新 update 校验           |
| ------- | ------------- | ---------------------- |
| 校验规则    | name 全局唯一     | lab_id + name 组合唯一     |

**重要知识点**

1. **业务不一致问题（重点）** 现在这套代码的规则是冲突的：
    - 新增：A实验室叫“电脑”、B实验室也叫“电脑” → 不允许，全局不能重名
    - 更新：只限制同一个实验室内部不能重名 ✅ 两种修正方向（二选一，业务对齐）
    - 方案1（统一为：全局name唯一）：update改成和create一样，只判断name，去掉lab_id条件
    - 方案2（统一为：同实验室name唯一）：create的校验也要加上lab_id，新增时校验 `lab_id+name` 组合唯一
        
        > 推荐业务上对齐，不然前端体验、规则理解会混乱。
        
2. 这两层都属于**业务层校验** 在service里先查库判断，提前给前端友好提示；属于业务兜底，数据库层面也可以加唯一索引做二次防护。

**思维误区**

❌ 误区1：新增和更新的校验天然就该不一样 
不是，是这里业务设计写得不一样；正常项目要统一一套唯一性规则。 
❌ 误区2：update里的lab_id是原来的实验室id 
不是，代码里lab_id是取payload里面更新后的lab_id，支持“设备挪到别的实验室”场景下的校验。 
❌ 误区3：新增也需要加 `id != xxx` 
新增的数据还没有入库，没有id，完全不需要。
